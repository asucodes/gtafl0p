+fn window_layout_line() {
+    if !crate::asked_for!("VICTORY_WINDOW_LAYOUT") {
+        return;
+    }
+    static LAST: AtomicU64 = AtomicU64::new(0);
+    let now = crate::host::now_ms();
+    let last = LAST.load(Ordering::Relaxed);
+    if last != 0 && now.saturating_sub(last) < 30_000 {
+        return;
+    }

+    let spans: Vec<_> = {
+        let map = LOOSE.read().unwrap_or_else(|e| e.into_inner());
+        if map.len() < 64 || LAST.compare_exchange(last, now.max(1), Ordering::Relaxed, Ordering::Relaxed).is_err() {
+            return;
+        }

+        map.values()
+            .filter(|s| seen.insert(s.id))
+            .map(|s| (s.start, s.len, s.in_place(), s.written.load(Ordering::Relaxed)))

+    let mut high_aligned = 0;
+    let mut high_written = 0;
+    let mut scattered = 0;
+    let mut holes = 0;
+    let mut no_first = 0;
+    let mut section_backed = 0;
+    let mut section_runs = BTreeMap::<usize, Vec<(u64, u64)>>::new();
+    for (start, len, placed, written) in spans {
+        if placed {
+            direct += len;
+            continue;
+        }
+        let (layout, backed) = crate::inplace::layout(start as usize, len as usize);
+        section_backed += backed;
+        match layout {
+            crate::inplace::Cover::Run(section, offset) => {
+                continuous += len;
+                if start >= 1 << 32 && offset.is_multiple_of(64 << 10) {
+                    high_aligned += len;
+                    if written {
+                        high_written += len;
+                    }
+                    section_runs.entry(section).or_default().push((offset, len));
+                }
+            }
+            crate::inplace::Cover::Scattered => scattered += len,
+            crate::inplace::Cover::Holes => holes += len,
+            crate::inplace::Cover::None => no_first += len,
+        }

+        &mut self,
+        first: usize,
+        last: usize,
+        query: &mut impl FnMut(usize, usize) -> Option<Written>,
+    ) -> bool {
+        if self.covered.covers(first, last) {
+            return true;
+        }
+        self.kernel_calls += 1;
+        let Some(answer) = query(first, last) else {
+            self.failures += 1;
+            return false;
+        };
+        self.covered.insert(first, last);
+        self.written.extend(answer);
+        true
+    }
+}
+
+fn collect(
+    mut ranges: Vec<(usize, usize)>,
+    mut region_at: impl FnMut(usize) -> Option<Region>,
+    mut query: impl FnMut(usize, usize) -> Option<Written>,
+) -> Batch {
+    ranges.sort_unstable();
+    ranges.dedup();
+    let mut batch = Batch::default();
+    let mut region: Option<Region> = None;
+    let mut i = 0;
+    while i < ranges.len() {
+        let (first, last) = ranges[i];
+        if region.is_none_or(|current| first >= current.end) {
+            batch.regions += 1;
+            region = region_at(first);
+        }
+        i += 1;
+        let Some(current) = region.filter(|current| last <= current.end) else {
+            batch.ask(first, last, &mut query);
+            continue;
+        };
+        if !current.private {
+            // Views are tracked by the existing guard path when refreshed.
+            // No successful kernel query means no coverage is published.
+            batch.deferred += 1;
+            continue;
+        }
+        let mut end = last;
+        let start = i - 1;
+        while let Some(&(next, next_end)) = ranges.get(i) {
+            if next > end + AHEAD_GAP
+                || next_end > current.end
+                || end.max(next_end) - first > AHEAD_MAX
+            {
+                break;
+            }
+            end = end.max(next_end);
+            i += 1;
+        }
+        // A region query is only a hint. A mapping can change, or private
+        // memory can have been allocated without MEM_WRITE_WATCH.
+        if !batch.ask(first, end, &mut query) && i > start + 1 {
+            for &(from, to) in &ranges[start..i] {
+                batch.ask(from, to, &mut query);
+            }
+        }
+    }
+    batch
+}

+pub(super) fn ask(ranges: &[(usize, usize)]) {

+    let ranges = ranges
+        .iter()

+        .map(|&(start, end)| (start / PAGE, end.div_ceil(PAGE)))
+        .collect();
+    let batch = collect(ranges, region_at, kernel_written);

+    }

+    // overlapping write-watch bits ahead of this answer.
