+/// Whether the threads the submitting thread waits for run at its priority
+/// (`VICTORY_HELPER_PRIORITY`, on with `1`; `VICTORY_AB` can flip it while
+/// the title runs): above normal, ahead of the title's threads and of the
+/// threads that translate lists. They are the threads that spread its work
+/// (`shadow::wide`, the pools of `land`), the one that makes the heaps'
+/// records and the one that passes over the heap's views beside the lists.
+///
+/// On a processor with six cores (i5-12400F, 12 threads) the title's threads
+/// and eight translating lists kept every core busy, and the submitting
+/// thread (above normal since `queue::behind`) stood waiting for helpers at
+/// normal priority that were not running. Measured with `VICTORY_AB` in one
+/// run (run10, Series S at 540p, medians of 5-second windows in the world,
+/// off / on): 8.6 / 9.6 fps, refreshes 23.1 / 17.6 ms a frame, waiting for
+/// the heaps' records 6.2 / 2.5 ms, the card idle behind the submitting
+/// thread 24.7 / 13.2 ms. Lowering the translating threads below normal as
+/// well made each list call about 5 % slower, with no clear gain of its own.
+fn helpers_raised() -> bool {
+    crate::asked_for!("VICTORY_HELPER_PRIORITY")
+}

+/// A helper's priority ([`helpers_raised`]). [`Self::keep`] is called before
+/// each piece of work; it asks the system only when the switch has changed.
+pub(crate) struct Priority {
+    applied: Option<bool>,
+}
+
+impl Priority {
+    pub(crate) fn new() -> Self {
+        let mut priority = Self { applied: None };
+        priority.keep();
+        priority
+    }
+
+    pub(crate) fn keep(&mut self) {
+        use windows_sys::Win32::System::Threading::{
+            GetCurrentThread, SetThreadPriority, THREAD_PRIORITY_ABOVE_NORMAL, THREAD_PRIORITY_NORMAL,
+        };
+        let on = helpers_raised();
+        if self.applied == Some(on) {
+            return;
+        }
+        let priority = if on { THREAD_PRIORITY_ABOVE_NORMAL } else { THREAD_PRIORITY_NORMAL };

+        unsafe { SetThreadPriority(GetCurrentThread(), priority) };

+enum AdapterPick {
+    /// Hardware adapter with the most dedicated video memory.
+    MostMemory,
+    /// `EnumAdapters1` index, software adapters included in the count.
+    Index(u32),
+    /// Case-insensitive part of the adapter name (`NVIDIA`).
+    Name(String),

+fn parse_adapter_pick(raw: Option<&str>) -> AdapterPick {
+    let Some(raw) = raw.map(str::trim).filter(|s| !s.is_empty()) else {
+        return AdapterPick::MostMemory;
+    };
+    match raw.parse::<u32>() {
+        Ok(index) => AdapterPick::Index(index),
+        Err(_) => AdapterPick::Name(raw.to_lowercase()),
+    }

+fn adapter_name(desc: &DXGI_ADAPTER_DESC1) -> String {
+    let len = desc.Description.iter().position(|&c| c == 0).unwrap_or(desc.Description.len());
+    String::from_utf16_lossy(&desc.Description[..len])

+/// The adapter the device is created on.
+///
+/// `VICTORY_D3D12_ADAPTER` picks one: a decimal index, or a case-insensitive
+/// part of the adapter name. Unset, the hardware adapter with the most
+/// dedicated video memory is used, so a laptop whose panel is on the
+/// integrated GPU does not run the title there. Software adapters are skipped
+/// except when an index names one.
+pub(crate) fn preferred_adapter() -> windows::core::Result<IDXGIAdapter1> {
+    // SAFETY: plain DXGI adapter queries.
+    unsafe {
+        let factory: IDXGIFactory4 = CreateDXGIFactory2(DXGI_CREATE_FACTORY_FLAGS(0))?;
+        let pick = match std::env::var("VICTORY_D3D12_ADAPTER") {
+            Ok(raw) => parse_adapter_pick(Some(&raw)),
+            Err(_) => AdapterPick::MostMemory,
+        };
+        let mut best: Option<(u64, IDXGIAdapter1)> = None;
+        for i in 0..16u32 {
+            let Ok(adapter) = factory.EnumAdapters1(i) else { break };
+            let desc = adapter.GetDesc1()?;
+            let software = desc.Flags & DXGI_ADAPTER_FLAG_SOFTWARE.0 as u32 != 0;
+            if software && !matches!(pick, AdapterPick::Index(_)) {
+                continue;
+            }
+            let name = adapter_name(&desc);
+            let dedicated = desc.DedicatedVideoMemory as u64;
+            let chosen = match &pick {
+                AdapterPick::Index(n) => i == *n,
+                AdapterPick::Name(part) => name.to_lowercase().contains(part),
+                AdapterPick::MostMemory => best.as_ref().is_none_or(|(mem, _)| dedicated > *mem),
+            };
+            if !chosen {
+                continue;
+            }
+            if !matches!(pick, AdapterPick::MostMemory) {
+                tracing::info!(adapter = %name, dedicated_mb = dedicated >> 20, "host adapter");
+                return Ok(adapter);
+            }
+            best = Some((dedicated, adapter));
+        }
+        let Some((_, adapter)) = best else {
+            return Err(windows::core::Error::from_hresult(windows::Win32::Foundation::E_FAIL));
+        };
+        let desc = adapter.GetDesc1()?;
+        tracing::info!(
+            adapter = %adapter_name(&desc),
+            dedicated_mb = desc.DedicatedVideoMemory >> 20,
+            "host adapter (most dedicated memory)"
+        );
+        Ok(adapter)
+    }
+}

+    /// made and freed: one section per allocation against 2 MiB sections
+    /// (fxceless/victory-fork#1089). Ignored: it maps 4 GiB and runs for a
+    /// minute. `cargo test --release -p victory-hle --lib physical_remap_bench
+    /// -- --ignored --nocapture`; `VICTORY_SECTION_BENCH=<MiB>,<seconds>,

+    #[test]
+    #[ignore = "a memory remap benchmark adapted from upstream PR #3177"]
+    fn physical_remap_bench() {
+        use std::sync::atomic::AtomicBool;
+        use std::time::{Duration, Instant};
+
+        let knobs: Vec<usize> = std::env::var("VICTORY_SECTION_BENCH")
+            .unwrap_or_default()
+            .split(',')
+            .filter_map(|v| v.trim().parse().ok())
+            .collect();
+        let knob = |i: usize, default: usize| knobs.get(i).copied().unwrap_or(default);
+        let (mib, seconds, rounds, pressure) =
+            (knob(0, 4096), knob(1, 10), knob(2, 3), knob(3, 2048));
+        let n = mib * MIB / PAGE;
+        let percentile = |v: &mut Vec<f64>, p: usize| {
+            v.sort_by(f64::total_cmp);
+            v[(v.len() * p / 100).min(v.len() - 1)]
+        };
+        let name = |mb: usize| {
+            if mb == 0 {
+                "one".to_string()
+            } else {
+                format!("{mb} MiB")
+            }
+        };
+        let mut results: Vec<(usize, Vec<[f64; 5]>)> = vec![(0, vec![]), (2, vec![])];
+        for round in 0..rounds {
+            for (mb, rows) in &mut results {
+                let pages = allocate_in(n, pages_per_section(*mb));
+                let at = XMemVirtualAlloc(
+                    std::ptr::null_mut(),
+                    n * PAGE,
+                    MEM_RESERVE | PAGES_64K,
+                    0,
+                    PAGE_READWRITE,
+                );
+                assert!(!at.is_null());
+                let base = at as usize;
+                // A view a page, each page touched, as the title maps its pool.
+                for (k, page) in pages.iter().enumerate() {
+                    let page_at = (base + k * PAGE) as *mut c_void;
+                    assert_eq!(XMemMapPhysicalPages(page_at, 1, page), page_at);
+                    for b in (0..PAGE).step_by(4096) {
+                        // SAFETY: mapped read-write just now.
+                        unsafe { page_at.cast::<u8>().byte_add(b).write(1) };
+                    }
+                }
+                let stop = AtomicBool::new(false);
+                let mut calls = Vec::new();
+                std::thread::scope(|scope| {
+                    // Another user of memory: made, touched and freed again.
+                    scope.spawn(|| {
+                        while pressure > 0 && !stop.load(Ordering::Relaxed) {
+                            let mut block = vec![0u8; pressure * MIB];
+                            for b in block.iter_mut().step_by(4096) {
+                                *b = 1;
+                            }
+                            std::hint::black_box(&block);
+                        }
+                    });
+                    let mut seed = 0x9e37_79b9_7f4a_7c15u64;
+                    let mut next = || {
+                        seed ^= seed << 13;
+                        seed ^= seed >> 7;
+                        seed ^= seed << 17;
+                        seed as usize % n
+                    };
+                    let start = Instant::now();
+                    while start.elapsed() < Duration::from_secs(seconds as u64) {
+                        let (k, page) = (next(), pages[next()]);
+                        let page_at = (base + k * PAGE) as *mut c_void;
+                        let t = Instant::now();
+                        assert_eq!(XMemMapPhysicalPages(page_at, 1, &page), page_at);
+                        calls.push(t.elapsed().as_secs_f64() * 1e6);
+                        // SAFETY: mapped read-write just now.
+                        unsafe { page_at.cast::<u8>().write(2) };
+                    }
+                    stop.store(true, Ordering::Relaxed);
+                });
+                let slow: Vec<f64> = calls.iter().copied().filter(|&us| us >= 1000.0).collect();
+                let row = [
+                    calls.len() as f64 / seconds as f64,
+                    percentile(&mut calls, 50),
+                    percentile(&mut calls, 99),
+                    slow.len() as f64,
+                    slow.iter().sum::<f64>() / 1000.0,
+                ];
+                println!(
+                    "round {round}, sections {}: {:.0} remaps/s, median {:.1} us, p99 {:.1} us, {} of 1 ms or more ({:.0} ms)",
+                    name(*mb),
+                    row[0],
+                    row[1],
+                    row[2],
+                    row[3],
+                    row[4]
+                );
+                rows.push(row);
+                assert_eq!(XMemMapPhysicalPages(at, n, std::ptr::null()), at);
+                assert_eq!(XMemFreePhysicalPages(n, pages.as_ptr()), 1);
+            }
+        }
+        for (mb, rows) in &results {
+            let median = |i: usize| {
+                let mut v: Vec<f64> = rows.iter().map(|r| r[i]).collect();
+                percentile(&mut v, 50)
+            };
+            println!(
+                "median of {rounds} rounds, sections {}: {:.0} remaps/s, median {:.1} us, p99 {:.1} us, {} of 1 ms or more ({:.0} ms)",
+                name(*mb),
+                median(0),
+                median(1),
+                median(2),
+                median(3),
+                median(4)
+            );
+        }
+    }
